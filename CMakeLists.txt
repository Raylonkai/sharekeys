cmake_minimum_required(VERSION 3.21)
project(ShareKeys VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHAREKEYS_BUILD_UI "Build the Qt host/client application" ON)
option(SHAREKEYS_BUILD_HEADLESS "Build the command-line host/client application" OFF)
include(CTest)

add_library(sharekeys_core src/core/control_engine.cpp src/core/monitor_layout.cpp)
target_include_directories(sharekeys_core PUBLIC src)
if(MSVC)
    target_compile_options(sharekeys_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(sharekeys_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_library(sharekeys_platform src/platform/windows/windows_backend.cpp)
    target_include_directories(sharekeys_platform PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(sharekeys_platform PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(sharekeys_platform PUBLIC sharekeys_core user32)
else()
    add_library(sharekeys_platform src/platform/unsupported_backend.cpp)
    target_link_libraries(sharekeys_platform PUBLIC sharekeys_core)
endif()

if(SHAREKEYS_BUILD_UI OR SHAREKEYS_BUILD_HEADLESS)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network)
    set(CMAKE_AUTOMOC ON)
    add_library(sharekeys_network src/network/protocol.cpp src/network/peer.cpp)
    target_link_libraries(sharekeys_network PUBLIC sharekeys_core Qt${QT_VERSION_MAJOR}::Network)
    add_library(sharekeys_runtime src/app/runtime.cpp)
    target_link_libraries(sharekeys_runtime PUBLIC sharekeys_network sharekeys_platform)
endif()

if(SHAREKEYS_BUILD_HEADLESS)
    add_executable(sharekeys-node src/cli/main.cpp)
    target_link_libraries(sharekeys-node PRIVATE sharekeys_runtime Qt${QT_VERSION_MAJOR}::Core)
    install(TARGETS sharekeys-node RUNTIME DESTINATION .)
endif()

if(SHAREKEYS_BUILD_UI)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    add_executable(sharekeys WIN32 src/ui/main.cpp src/ui/main_window.cpp)
    target_link_libraries(sharekeys PRIVATE sharekeys_runtime Qt${QT_VERSION_MAJOR}::Widgets)
    install(TARGETS sharekeys RUNTIME DESTINATION .)
    install(DIRECTORY docs DESTINATION .)
    install(FILES README.md DESTINATION .)
    install(DIRECTORY third_party/deskflow DESTINATION licenses)
    if(BUILD_TESTING)
        find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
        add_executable(network_tests tests/network_tests.cpp)
        target_link_libraries(network_tests PRIVATE sharekeys_runtime Qt${QT_VERSION_MAJOR}::Test)
        add_test(NAME network COMMAND network_tests)
        add_test(NAME ui_smoke COMMAND sharekeys --smoke-test)
        add_test(NAME ui_demo_smoke COMMAND sharekeys --demo --smoke-test)
        set_tests_properties(ui_smoke ui_demo_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    endif()
endif()
if(SHAREKEYS_BUILD_HEADLESS AND BUILD_TESTING AND NOT SHAREKEYS_BUILD_UI)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(network_tests tests/network_tests.cpp)
    target_link_libraries(network_tests PRIVATE sharekeys_runtime Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME network COMMAND network_tests)
endif()
if(BUILD_TESTING)
    add_executable(control_engine_tests tests/control_engine_tests.cpp)
    target_link_libraries(control_engine_tests PRIVATE sharekeys_core)
    add_test(NAME control_engine COMMAND control_engine_tests)
    add_executable(layout_tests tests/layout_tests.cpp)
    target_link_libraries(layout_tests PRIVATE sharekeys_core)
    add_test(NAME layout COMMAND layout_tests)
endif()
